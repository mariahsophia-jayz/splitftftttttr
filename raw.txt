local function Root()
	local character = LocalPlayer.Character
	return character and character:FindFirstChild("HumanoidRootPart")
end

local function Humanoid()
	local character = LocalPlayer.Character
	return character and character:FindFirstChildOfClass("Humanoid")
end

local function Alive()
	local humanoid = Humanoid()
	return humanoid ~= nil and humanoid.Health > 0
end

local function Level()
	local data = LocalPlayer:FindFirstChild("Data")
	local value = data and data:FindFirstChild("Level")
	return value and value.Value or 1
end

local TravelTarget

local function HaltTravel()
	TravelTarget = nil
end

local function TravelTo(place)
	TravelTarget = place
end

local function TravelToPoint(position)
	TravelTarget = CFrame.new(position)
end

local function HeldTool()
	local character = LocalPlayer.Character
	return character and character:FindFirstChildOfClass("Tool")
end

local function Equip(kind)
	local character = LocalPlayer.Character
	if not character then
		return
	end
	local held = character:FindFirstChildOfClass("Tool")
	if held and held.ToolTip == kind then
		return
	end
	for _, tool in pairs(Backpack:GetChildren()) do
		if tool:IsA("Tool") and tool.ToolTip == kind then
			character.Humanoid:EquipTool(tool)
			return
		end
	end
end

local function Swing()
	local tool = HeldTool()
	if tool then
		tool:Activate()
	end
end

local function RefreshAttack()
	if not CombatFramework then
		return
	end
	local controller = CombatFramework.activeController
	if controller then
		controller.attackDelay = 0
		controller.attackTime = 0.3
		controller.hitboxMagnitude = 70
	end
end

local function Mobs(range)
	local root = Root()
	local found = {}
	if not root then
		return found
	end
	local enemies = workspace:FindFirstChild("Enemies")
	if not enemies then
		return found
	end
	for _, mob in pairs(enemies:GetChildren()) do
		local part = mob:FindFirstChild("HumanoidRootPart")
		local humanoid = mob:FindFirstChildOfClass("Humanoid")
		if part and humanoid and humanoid.Health > 0 then
			local distance = (part.Position - root.Position).Magnitude
			if distance <= range then
				table.insert(found, { model = mob, part = part, humanoid = humanoid, distance = distance })
			end
		end
	end
	return found
end

local function NearestMob(range)
	local best
	for _, mob in ipairs(Mobs(range)) do
		if not best or mob.distance < best.distance then
			best = mob
		end
	end
	return best
end

local function FindMob(name)
	local enemies = workspace:FindFirstChild("Enemies")
	if not enemies then
		return
	end
	for _, mob in pairs(enemies:GetChildren()) do
		local part = mob:FindFirstChild("HumanoidRootPart")
		local humanoid = mob:FindFirstChildOfClass("Humanoid")
		if mob.Name == name and part and humanoid and humanoid.Health > 0 then
			return { model = mob, part = part, humanoid = humanoid, distance = 0 }
		end
	end
end

local function FaceMob(mob)
	local root = Root()
	if root and mob then
		root.CFrame = CFrame.lookAt(root.Position, mob.part.Position)
	end
end

local tetherTick = 0

local function Tether(range)
	local root = Root()
	if not root then
		return
	end
	tetherTick = tetherTick + 1
	local anchor = root.CFrame * CFrame.new(0, 0, -4)
	for _, mob in ipairs(Mobs(range)) do
		if tetherTick % 20 == 0 then
			pcall(function()
				mob.part:SetNetworkOwner(LocalPlayer)
			end)
		end
		mob.part.CFrame = anchor
	end
end

local function QuestTitle()
	local main = PlayerGui:FindFirstChild("Main")
	local quest = main and main:FindFirstChild("Quest")
	if not quest or not quest.Visible then
		return
	end
	local ok, title = pcall(function()
		return quest.Container.QuestTitle.Title.Text
	end)
	if ok then
		return title
	end
end

local function QuestRunning(monster)
	local title = QuestTitle()
	return title ~= nil and string.find(title, monster, 1, true) ~= nil
end

local Elements = {}

local function ItemPart(instance)
	if instance:IsA("BasePart") then
		return instance
	end
	if instance:IsA("Model") then
		return instance.PrimaryPart or instance:FindFirstChildWhichIsA("BasePart")
	end
end

local function MarkEsp(instance, color, label)
	if EspMarks[instance] or not instance.Parent then
		return
	end
	local highlight = Instance.new("Highlight")
	highlight.Name = "LunarScriptzFill"
	highlight.FillColor = color
	highlight.OutlineColor = color
	highlight.FillTransparency = 0.62
	highlight.OutlineTransparency = 0.1
	highlight.Adornee = instance
	highlight.Parent = EspFolder

	local billboard = Instance.new("BillboardGui")
	billboard.Name = "LunarScriptzLabel"
	billboard.Size = UDim2.fromOffset(160, 26)
	billboard.StudsOffsetWorldSpace = Vector3.new(0, 4, 0)
	billboard.AlwaysOnTop = true
	billboard.Adornee = instance
	billboard.Parent = EspFolder

	local text = Instance.new("TextLabel")
	text.Size = UDim2.fromScale(1, 1)
	text.BackgroundTransparency = 1
	text.Font = Enum.Font.GothamBold
	text.TextScaled = true
	text.TextColor3 = color
	text.TextStrokeTransparency = 0.5
	text.Text = label
	text.Parent = billboard

	EspMarks[instance] = { highlight = highlight, billboard = billboard, text = text, label = label, adornee = ItemPart(instance) or instance }
end

local function ClearEsp()
	for instance, mark in pairs(EspMarks) do
		mark.highlight:Destroy()
		mark.billboard:Destroy()
		EspMarks[instance] = nil
	end
end

local SeaPatterns = { "Terror", "Sea Beast", "Shark", "Piranha", "Fish Crew", "Kitsune" }

local function IsSeaMob(name)
	for _, pattern in ipairs(SeaPatterns) do
		if string.find(name, pattern, 1, true) then
			return true
		end
	end
	return false
end

local function NearestItem(pattern, range, container)
	local root = Root()
	if not root then
		return
	end
	local best, bestDistance
	for _, instance in pairs((container or workspace):GetChildren()) do
		if string.find(instance.Name, pattern, 1, true) then
			local part = ItemPart(instance)
			if part then
				local distance = (part.Position - root.Position).Magnitude
				if distance <= range and (not bestDistance or distance < bestDistance) then
					best, bestDistance = part, distance
				end
			end
		end
	end
	return best
end

local function PickQuest()
	local level = Level()
	for _, entry in ipairs(QuestLine) do
		if level >= entry[1] and level <= entry[2] then
			return entry
		end
	end
end

local questCooldown = 0

local function FarmTick()
	local entry = PickQuest()
	if not entry then
		HaltTravel()
		return
	end
	if not QuestRunning(entry[3]) and os.clock() > questCooldown then
		questCooldown = os.clock() + 1
		if QuestTitle() then
			CommF:InvokeServer("AbandonQuest")
		end
		ClaimQuest(entry)
	end
	local mob = FindMob(entry[3])
	if mob then
		TravelTo(CFrame.new(mob.part.Position))
	else
		TravelToPoint(entry[6])
	end
end

local fishCooldown = 0

local function CastPoint()
	local root = Root()
	if not root then
		return
	end
	local origin = root.Position + root.CFrame.LookVector * 6
	local params = RaycastParams.new()
	params.FilterType = Enum.RaycastFilterType.Exclude
	params.FilterDescendantsInstances = { LocalPlayer.Character, workspace:FindFirstChild("Enemies") }
	local hit = workspace:Raycast(origin, root.CFrame.LookVector * 140, params)
	local point = hit and hit.Position or (origin + root.CFrame.LookVector * 60)
	local water
	if WaterHeight then
		local ok, height = pcall(WaterHeight, point)
		water = ok and height or nil
	end
	return Vector3.new(point.X, math.max(point.Y, water or point.Y), point.Z)
end

local function FishTick()
	local character = LocalPlayer.Character
	if not character or not Alive() or os.clock() < fishCooldown then
		return
	end
	local rod = character:FindFirstChild(Flags.rod)
	if not rod then
		local spare = Backpack:FindFirstChild(Flags.rod)
		if spare then
			character.Humanoid:EquipTool(spare)
		end
		return
	end
	local state = rod:GetAttribute("State")
	if state == "ReeledIn" or state == nil then
		CommF:InvokeServer("StartCasting")
		task.wait(0.15)
		CommF:InvokeServer("CastLineAtLocation", CastPoint(), 100, true)
		fishCooldown = os.clock() + 0.6
	elseif state == "Biting" or rod:GetAttribute("ServerState") == "Biting" then
		CommF:InvokeServer("Catching", true)
		task.wait(0.1)
		CommF:InvokeServer("Catch", 1)
		fishCooldown = os.clock() + 0.4
	end
end

local raidIsland = 1
local raidTimer = 0
local raidSummonCooldown = 0

local function RaidSummoner()
	local map = workspace:FindFirstChild("Map")
	if not map then
		return
	end
	for _, name in ipairs({ "Boat Castle", "CircleIsland" }) do
		local region = map:FindFirstChild(name)
		local summon = region and (region:FindFirstChild("RaidSummon2") or region:FindFirstChild("RaidSummon"))
		local button = summon and summon:FindFirstChild("Button")
		local main = button and button:FindFirstChild("Main")
		local detector = main and main:FindFirstChild("ClickDetector")
		if detector then
			return detector
		end
	end
end

local function RaidSummonTick()
	local detector = RaidSummoner()
	if not detector then
		return
	end
	local part = ItemPart(detector.Parent)
	local root = Root()
	if not part or not root then
		return
	end
	if (part.Position - root.Position).Magnitude > 12 then
		TravelTo(CFrame.new(part.Position))
		return
	end
	if os.clock() < raidSummonCooldown then
		return
	end
	raidSummonCooldown = os.clock() + 3
	if fireclickdetector then
		fireclickdetector(detector)
	end
end

local function RaidIsland()
	local origin = workspace:FindFirstChild("_WorldOrigin")
	local locations = origin and origin:FindFirstChild("Locations")
	if not locations then
		return
	end
	local island = locations:FindFirstChild("Island " .. raidIsland)
	if island then
		return island
	end
end

local function RaidTick()
	local island = RaidIsland()
	if not island then
		raidIsland = 1
		return
	end
	local mob = NearestMob(400)
	if mob then
		FaceMob(mob)
		Swing()
		return
	end
	if os.clock() < raidTimer then
		TravelTo(CFrame.new(island.Position))
		return
	end
	raidIsland = raidIsland % 5 + 1
	raidTimer = os.clock() + 4
end

local safeNotified = false

local function SafeModeTick()
	local humanoid = Humanoid()
	if not humanoid then
		return
	end
	if humanoid.Health <= humanoid.MaxHealth * 0.3 then
		if Flags.questFarm or Flags.bossFarm or Flags.materialFarm then
			Flags.questFarm = false
			Flags.bossFarm = false
			Flags.materialFarm = false
			Flags.mobAura = false
			HaltTravel()
			for _, name in ipairs({ "questFarm", "bossFarm", "materialFarm", "mobAura" }) do
				if Elements[name] then
					Elements[name]:Set(false)
				end
			end
			if not safeNotified then
				safeNotified = true
				Notify("Safe Mode", "Health low, farming paused", "shield-alert")
			end
		end
	else
		safeNotified = false
	end
end

local function InventoryNames(pattern)
	local ok, inventory = pcall(function()
		return CommF:InvokeServer("getInventory")
	end)
	local names = {}
	if ok and type(inventory) == "table" then
		for name in pairs(inventory) do
			if string.find(name, pattern, 1, true) then
				table.insert(names, name)
			end
		end
	end
	table.sort(names)
	return names
end

end

local function AzureTick()
	local part = NearestItem("Azure", 5000)
	if part then
		TravelTo(CFrame.new(part.Position))
	end
end

local function GolemTick()
	local target
	for _, mob in ipairs(Mobs(3000)) do
		if string.find(mob.model.Name, "Golem", 1, true) and (not target or mob.distance < target.distance) then
			target = mob
		end
	end
	if target then
		TravelTo(CFrame.new(target.part.Position))
	end
end

local function SyncChestEsp()
	for _, instance in pairs(workspace:GetChildren()) do
		if string.find(instance.Name, "Chest", 1, true) and ItemPart(instance) then
			MarkEsp(instance, EspColors.Chest, "Chest")
		end
	end
end

local function SyncFruitEsp()
	for _, instance in pairs(workspace:GetChildren()) do
		if string.find(instance.Name, "Fruit", 1, true) then
			MarkEsp(instance, EspColors.Fruit, instance.Name)
		end
	end
	local fruits = workspace:FindFirstChild("Fruits")
	if fruits then
		for _, instance in pairs(fruits:GetChildren()) do
			MarkEsp(instance, EspColors.Fruit, instance.Name)
		end
	end
end

local function SyncBerryEsp()
	for _, instance in pairs(CollectionService:GetTagged("BerryBush")) do
		MarkEsp(instance, EspColors.Berry, "Berry")
	end
end

local function SyncPlayerEsp()
	for _, player in pairs(Players:GetPlayers()) do
		if player ~= LocalPlayer and player.Character then
			MarkEsp(player.Character, EspColors.Player, player.Name)
		end
	end
end

local function SyncBeastEsp()
	local enemies = workspace:FindFirstChild("Enemies")
	if not enemies then
		return
	end
	for _, mob in pairs(enemies:GetChildren()) do
		if IsSeaMob(mob.Name) then
			MarkEsp(mob, EspColors.Beast, mob.Name)
		end
	end
end

local function SyncEsp()
	if Flags.chestEsp then
		SyncChestEsp()
	end
	if Flags.fruitEsp then
		SyncFruitEsp()
	end
	if Flags.berryEsp then
		SyncBerryEsp()
	end
	if Flags.playerEsp then
		SyncPlayerEsp()
	end
	if Flags.beastEsp then
		SyncBeastEsp()
	end
	local root = Root()
	for instance, mark in pairs(EspMarks) do
		if not instance.Parent then
			mark.highlight:Destroy()
			mark.billboard:Destroy()
			EspMarks[instance] = nil
		elseif root then
			local ok, position = pcall(function()
				return mark.adornee.Position
			end)
			if ok then
				mark.text.Text = string.format("%s [%dm]", mark.label, (position - root.Position).Magnitude)
			end
		end
	end
end

local guitarCooldown = 0

local function Beat()
	MovementTick()
	Equip(Flags.weapon)
	if Flags.fastAttack then
		RefreshAttack()
	end
	if Flags.bringMob then
		Tether(Flags.auraRange)
	end
	if Flags.questFarm then
		FarmTick()
	end
	if Flags.bossFarm then
		BossTick()
	end
	if Flags.swordFarm then
		SwordTick()
	end
	if Flags.materialFarm then
		MaterialTick()
	end
	if Flags.mobAura or Flags.dojoAura then
		AuraTick()
	end
	if Flags.chestFarm then

task.spawn(function()
	while true do
		RunService.Heartbeat:Wait()
		local root = Root()
		if root and TravelTarget and Alive() then
			local offset = TravelTarget.Position - root.Position
			local distance = offset.Magnitude
			if distance > 3 then
				local step = math.clamp(distance * 0.35, 5, 400)
				root.CFrame = CFrame.lookAt(root.Position + offset.Unit * step, TravelTarget.Position)
				root.AssemblyLinearVelocity = Vector3.zero
			end
		end
	end
end)
